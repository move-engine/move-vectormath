102 template <typename T>
103 [[nodiscard]] inline std::optional<Point3<T>> TryProjectPoint(
104 const Mat4<T>& worldToClip, const Point3<T>& worldPoint) noexcept
105 {
106 // Homogeneous projection follows the OpenGL/Direct3D pipeline and is
107 // cross-checked against GLM project/unProject and DirectXMath
108 // XMVector3Project/Unproject (both MIT). W failure is explicit.
109 const Vec4<T> clip =
110 Vec4<T>(worldPoint.X(), worldPoint.Y(), worldPoint.Z(), T(1)) *
111 worldToClip;
112 if (clip.W() == T(0) || !std::isfinite(clip.W()))
113 {
114 return std::nullopt;
115 }
116 const Point3<T> result(clip.X() / clip.W(), clip.Y() / clip.W(),
117 clip.Z() / clip.W());
118 return std::isfinite(result.X()) && std::isfinite(result.Y()) &&
119 std::isfinite(result.Z())
120 ? std::optional<Point3<T>>(result)
121 : std::nullopt;
122 }

140 std::isfinite(result.Z())

156 template <typename T>
157 [[nodiscard]] inline std::optional<Point3<T>> TryProjectToViewport(
158 const Mat4<T>& worldToClip,
159 const Point3<T>& worldPoint,
160 const Viewport<T>& viewport,
161 ClipDepth clipDepth,
162 ViewportYAxis yAxis = ViewportYAxis::Down) noexcept
163 {
164 const auto ndc = TryProjectPoint(worldToClip, worldPoint);
165 if (!ndc)
166 {
167 return std::nullopt;
168 }
169 const T normalizedX = ndc->X() * T(0.5) + T(0.5);
170 const T normalizedY = yAxis == ViewportYAxis::Down
171 ? T(0.5) - ndc->Y() * T(0.5)
172 : ndc->Y() * T(0.5) + T(0.5);
173 const T normalizedDepth = clipDepth == ClipDepth::ZeroToOne
174 ? ndc->Z()
175 : ndc->Z() * T(0.5) + T(0.5);
176 return Point3<T>(viewport.X() + normalizedX * viewport.Width(),
177 viewport.Y() + normalizedY * viewport.Height(),
178 viewport.MinimumDepth() +
179 normalizedDepth * (viewport.MaximumDepth() -
180 viewport.MinimumDepth()));
181 }

183 template <typename T>
184 [[nodiscard]] inline std::optional<Point3<T>>
185 TryUnprojectFromViewportWithInverse(
186 const Mat4<T>& clipToWorld,
187 const Point3<T>& viewportPoint,
188 const Viewport<T>& viewport,
189 ClipDepth clipDepth,
190 ViewportYAxis yAxis = ViewportYAxis::Down) noexcept
191 {
192 const T normalizedX =
193 (viewportPoint.X() - viewport.X()) / viewport.Width();
194 const T normalizedY =
195 (viewportPoint.Y() - viewport.Y()) / viewport.Height();
196 const T normalizedDepth =
197 (viewportPoint.Z() - viewport.MinimumDepth()) /
198 (viewport.MaximumDepth() - viewport.MinimumDepth());
199 const T ndcX = normalizedX * T(2) - T(1);
200 const T ndcY = yAxis == ViewportYAxis::Down ? T(1) - normalizedY * T(2)
201 : normalizedY * T(2) - T(1);
202 const T ndcZ = clipDepth == ClipDepth::ZeroToOne
203 ? normalizedDepth
204 : normalizedDepth * T(2) - T(1);
205 return TryUnprojectPointWithInverse(clipToWorld,
206 Point3<T>(ndcX, ndcY, ndcZ));
207 }

210 [[nodiscard]] inline std::optional<Ray3<T>> TryMakeViewportRay(
211 const Mat4<T>& clipToWorld,
212 const Vec2<T>& viewportPosition,
213 const Viewport<T>& viewport,
214 ClipDepth clipDepth,
215 DepthDirection depthDirection,
216 ViewportYAxis yAxis = ViewportYAxis::Down) noexcept
217 {
218 const T nearDepth = depthDirection == DepthDirection::Forward
219 ? viewport.MinimumDepth()
220 : viewport.MaximumDepth();
221 const T farDepth = depthDirection == DepthDirection::Forward
222 ? viewport.MaximumDepth()
223 : viewport.MinimumDepth();
224 const T sampleDepth = nearDepth + (farDepth - nearDepth) * T(0.5);
225 const auto nearPoint = TryUnprojectFromViewportWithInverse(
226 clipToWorld,
227 Point3<T>(viewportPosition.X(), viewportPosition.Y(), nearDepth),
228 viewport, clipDepth, yAxis);
229 const auto samplePoint = TryUnprojectFromViewportWithInverse(
230 clipToWorld,
231 Point3<T>(viewportPosition.X(), viewportPosition.Y(), sampleDepth),
232 viewport, clipDepth, yAxis);
233 if (!nearPoint || !samplePoint)
234 {
235 return std::nullopt;
236 }
237 const auto direction =
238 Direction3<T>::TryFrom(*samplePoint - *nearPoint);
239 return direction
240 ? Ray3<T>::TryFromOriginDirection(*nearPoint, *direction)
241 : std::nullopt;
242 }
