71 template <typename T, typename RadiusForPlane>
72 [[nodiscard]] inline CullingResult ClassifyAgainstFrustum(
73 const Frustum3<T>& frustum,
74 const Point3<T>& center,
75 FrustumPlaneMask requestedPlanes,
76 T tolerance,
77 RadiusForPlane&& radiusForPlane) noexcept
78 {
79 FrustumPlaneMask remaining = FrustumPlaneMask::None;
80 const FrustumPlaneMask active =
81 requestedPlanes & frustum.ActivePlanes();
82 for (std::size_t index = 0U; index < 6U; ++index)
83 {
84 const FrustumPlane plane = static_cast<FrustumPlane>(index);
85 if (!HasPlane(active, plane))
86 {
87 continue;
88 }
89 const T radius = radiusForPlane(plane);
90 const T distance = frustum.Plane(plane).SignedDistance(center);
91 if (distance < -radius - tolerance)
92 {
93 return {Containment::Outside, FrustumPlaneMask::None};
94 }
95 if (distance <= radius + tolerance)
96 {
97 remaining |= PlaneMask(plane);
98 }
99 }
100 return {remaining == FrustumPlaneMask::None
101 ? Containment::Inside
102 : Containment::Intersecting,
103 remaining};
104 }
