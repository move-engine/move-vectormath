104 template <typename T>
105 [[nodiscard]] inline TrsDecomposition3<T> DecomposeTrs(
106 const AffineTransform3<T>& transform,
107 T orthogonalityTolerance = static_cast<T>(1.0e-5)) noexcept
108 {
109 const Vec3<T>& axisX = transform.AxisX();
110 const Vec3<T>& axisY = transform.AxisY();
111 const Vec3<T>& axisZ = transform.AxisZ();
112 T scaleX = Length(axisX);
113 T scaleY = Length(axisY);
114 T scaleZ = Length(axisZ);
115 if (!(scaleX > T(0)) || !(scaleY > T(0)) || !(scaleZ > T(0)))
116 {
117 return {TrsDecompositionStatus::Singular, std::nullopt,
118 Vec3<T>::Zero()};
119 }
120 if (!std::isfinite(scaleX) || !std::isfinite(scaleY) ||
121 !std::isfinite(scaleZ))
122 {
123 return {TrsDecompositionStatus::NumericalFailure, std::nullopt,
124 Vec3<T>::Zero()};
125 }
126
127 Vec3<T> normalizedX = axisX / scaleX;
128 const Vec3<T> normalizedY = axisY / scaleY;
129 const Vec3<T> normalizedZ = axisZ / scaleZ;
130 const Vec3<T> shear(Dot(normalizedX, normalizedY),
131 Dot(normalizedX, normalizedZ),
132 Dot(normalizedY, normalizedZ));
133 const T tolerance = std::abs(orthogonalityTolerance);
134 if (std::abs(shear.X()) > tolerance ||
135 std::abs(shear.Y()) > tolerance || std::abs(shear.Z()) > tolerance)
136 {
137 return {TrsDecompositionStatus::Sheared, std::nullopt, shear};
138 }
139
140 // A unit quaternion represents only proper rotations. Preserve an
141 // affine reflection by assigning its sign to the X scale before
142 // extracting the proper orthonormal basis. This is the deterministic
143 // Move policy; equivalent decompositions with another negative axis
144 // are intentionally not produced.
145 if (Dot(Cross(normalizedX, normalizedY), normalizedZ) < T(0))
146 {
147 normalizedX = -normalizedX;
148 scaleX = -scaleX;
149 }
150
151 const auto directionX = Direction3<T>::TryFrom(normalizedX);
152 const auto directionY = Direction3<T>::TryFrom(normalizedY);
153 const auto directionZ = Direction3<T>::TryFrom(normalizedZ);
154 if (!directionX || !directionY || !directionZ)
155 {
156 return {TrsDecompositionStatus::NumericalFailure, std::nullopt,
157 shear};
158 }
159 const auto rotation = Rotation3<T>::TryFromOrthonormalAxes(
160 *directionX, *directionY, *directionZ, tolerance);
161 if (!rotation)
162 {
163 return {TrsDecompositionStatus::NumericalFailure, std::nullopt,
164 shear};
165 }
166
167 return {TrsDecompositionStatus::Success,
168 TrsTransform3<T>(transform.Translation(), *rotation,
169 Vec3<T>(scaleX, scaleY, scaleZ)),
170 shear};
171 }
