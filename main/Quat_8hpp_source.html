122 [[nodiscard]] inline std::optional<Quat<T>> TryNormalize(
123 const Quat<T>& value) noexcept
124 {
125 const T maximum = std::max({std::abs(value.X()), std::abs(value.Y()),
126 std::abs(value.Z()), std::abs(value.W())});
127 if (!(maximum > T(0)) || !std::isfinite(maximum))
128 {
129 return std::nullopt;
130 }
131 const Quat<T> scaled = value / maximum;
132 const T lengthSquared = LengthSquared(scaled);
133 if (!(lengthSquared > T(0)) || !std::isfinite(lengthSquared))
134 {
135 return std::nullopt;
136 }
137 const T reciprocalLength = T(1) / std::sqrt(lengthSquared);
138 const Quat<T> result = scaled * reciprocalLength;
139 return std::isfinite(result.X()) && std::isfinite(result.Y()) &&
140 std::isfinite(result.Z()) && std::isfinite(result.W())
141 ? std::optional<Quat<T>>(result)

145 template <typename T>
146 [[nodiscard]] inline std::optional<Quat<T>> TryInverse(
147 const Quat<T>& value) noexcept
148 {
149 const T maximum = std::max({std::abs(value.X()), std::abs(value.Y()),
150 std::abs(value.Z()), std::abs(value.W())});
151 if (!(maximum > T(0)) || !std::isfinite(maximum))
152 {
153 return std::nullopt;
154 }
155
156 // Scaling before the squared norm follows LAPACK's overflow-avoiding
157 // xLASSQ pattern; dividing by maximum last also preserves invertible
158 // subnormal and near-maximum finite quaternions.
159 const Quat<T> scaled = value / maximum;
160 const T scaledLengthSquared = LengthSquared(scaled);
161 if (!(scaledLengthSquared > T(0)) ||
162 !std::isfinite(scaledLengthSquared))
163 {
164 return std::nullopt;
165 }
166 const Quat<T> result =
167 Quat<T>(-scaled.X(), -scaled.Y(), -scaled.Z(), scaled.W()) /
168 scaledLengthSquared / maximum;
169 return std::isfinite(result.X()) && std::isfinite(result.Y()) &&
170 std::isfinite(result.Z()) && std::isfinite(result.W())
171 ? std::optional<Quat<T>>(result)
172 : std::nullopt;
173 }
